```sql
CREATE TABLE article_entities (
      article_id INTEGER NOT NULL REFERENCES articles(id),
      kind TEXT NOT NULL CHECK(kind IN ('party','minister','state','city','topic')),
      slug TEXT NOT NULL,
      PRIMARY KEY (article_id, kind, slug)
    );
CREATE TABLE articles (
    id INTEGER PRIMARY KEY, -- Preserves the unique sequential ID from the scraper
    cluster_id TEXT,
    source_id INTEGER,
    title TEXT,
    url TEXT UNIQUE,
    content BLOB, -- compressed original full text (zlib compressed)
    image_url TEXT,
    scraped_at INTEGER, -- Unix timestamp (integer)
    category TEXT,
    sentiment TEXT,
    sentiment_target TEXT,
    rephrased_article BLOB, -- compressed rephrased summary (zlib compressed)
    party_mentioned TEXT, -- JSON array string
    ministers_mentioned TEXT, -- JSON array string
    states_mentioned TEXT, -- JSON array string
    cities_mentioned TEXT, -- JSON array string
    topic_tags TEXT, -- JSON array string
    civic_flag INTEGER DEFAULT 0, -- 1 for True, 0 for False
    civic_flag_score INTEGER DEFAULT 0,
    civic_flag_category TEXT,
    civic_flag_reason TEXT,
    classified_at INTEGER, -- Unix timestamp (integer)
    status TEXT DEFAULT 'scraped', rephrased_title TEXT, headline_verified INTEGER DEFAULT 0, translated_hi INTEGER DEFAULT 0, geo_focus TEXT, -- 'scraped', 'rephrased', 'classified'
    FOREIGN KEY(source_id) REFERENCES sources(id)
);
CREATE TABLE dead_endpoints (
           endpoint    TEXT PRIMARY KEY,
           status_code INTEGER,
           died_at     INTEGER NOT NULL
         );
CREATE TABLE event_articles (
      event_id   INTEGER NOT NULL REFERENCES events(id),
      article_id INTEGER NOT NULL REFERENCES articles(id),
      milestone  TEXT,
      event_date INTEGER,
      PRIMARY KEY (event_id, article_id)
    );
CREATE TABLE events (
      id            INTEGER PRIMARY KEY AUTOINCREMENT,
      title         TEXT,
      slug          TEXT UNIQUE,
      entity_keys   TEXT NOT NULL,
      centroid      BLOB NOT NULL,
      saga_id       INTEGER,
      first_seen    INTEGER, 
      last_seen     INTEGER,
      article_count INTEGER DEFAULT 1,
      state         TEXT DEFAULT 'open'
    , scope TEXT, translated_hi INTEGER DEFAULT 0);
CREATE TABLE indexnow_urls (
           url          TEXT PRIMARY KEY,
           lastmod      INTEGER NOT NULL,
           submitted_at INTEGER NOT NULL
         );
CREATE TABLE notification_logs (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        title TEXT NOT NULL,
        body TEXT,
        url TEXT,
        sent_count INTEGER DEFAULT 0,
        failed_count INTEGER DEFAULT 0,
        purged_count INTEGER DEFAULT 0,
        created_at INTEGER NOT NULL
    );
CREATE TABLE push_subscriptions (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        endpoint TEXT UNIQUE NOT NULL,
        p256dh TEXT NOT NULL,
        auth TEXT NOT NULL,
        user_agent TEXT,
        created_at INTEGER NOT NULL,
        last_active INTEGER NOT NULL
    );
CREATE TABLE sources (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT UNIQUE
);
CREATE TABLE timeline_checkpoint (
      id INTEGER PRIMARY KEY CHECK (id = 1),
      last_article_id INTEGER NOT NULL
    );
CREATE INDEX idx_ae_kind_slug ON article_entities(kind, slug, article_id);
CREATE INDEX idx_ae_lookup_desc ON article_entities(kind, slug, article_id DESC);
CREATE INDEX idx_articles_cat_scraped_desc ON articles(category, scraped_at DESC) WHERE status IN ('classified', 'entity_processed', 'processed');
CREATE INDEX idx_articles_cluster ON articles(cluster_id);
CREATE INDEX idx_articles_scraped ON articles(scraped_at);
CREATE INDEX idx_articles_scraped_category ON articles(scraped_at, category) WHERE status IN ('classified', 'entity_processed', 'processed');
CREATE INDEX idx_articles_source_id ON articles(source_id);
CREATE INDEX idx_articles_source_scraped ON articles(source_id, scraped_at DESC) WHERE status IN ('classified', 'entity_processed', 'processed');
CREATE INDEX idx_articles_status ON articles(status);
CREATE INDEX idx_articles_status_category_scraped ON articles(status, category, scraped_at);
CREATE INDEX idx_articles_status_civic_scraped ON articles(status, civic_flag, scraped_at);
CREATE INDEX idx_articles_status_scraped ON articles(status, scraped_at);
CREATE INDEX idx_articles_trans_hi ON articles (translated_hi, id DESC);
CREATE INDEX idx_ea_article ON event_articles(article_id);
CREATE INDEX idx_ea_event_date ON event_articles (event_id, event_date);
CREATE INDEX idx_events_last_seen ON events (last_seen);
CREATE INDEX idx_events_state_seen ON events(state, last_seen);
CREATE INDEX idx_events_trans_hi ON events (translated_hi, id DESC);
CREATE INDEX idx_notif_logs_created ON notification_logs(created_at);
CREATE INDEX idx_push_subs_endpoint ON push_subscriptions(endpoint);
```
